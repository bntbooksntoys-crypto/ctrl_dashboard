// My Websites — Google Apps Script backend (version 2: with Run on all devices) // Stores the site list in this Google Sheet, checks both passwords on Google's // servers, and carries the "Run" command that every open dashboard obeys. var SHEET_NAME = 'sites'; var VIEW_PW = '5442252'; // password to see the list var ADMIN_PW = '5905018'; // password to add, remove, or run sites function getSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['name', 'siteUrl', 'repoUrl', 'desc']); sh.appendRow(['Qmaz Solutions', 'https://bntbooksntoys-crypto.github.io/qmaz-solutions/', 'https://github.com/bntbooksntoys-crypto/qmaz-solutions', 'EdTech platform — learn in English, Tamil, Hindi and more languages.']); sh.appendRow(['Bntbooksandtoys', 'https://bntbooksntoys-crypto.github.io/bntbooksandtoys/', 'https://github.com/bntbooksntoys-crypto/bntbooksandtoys', '']); } return sh; } function getControl_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName('control'); if (!sh) { sh = ss.insertSheet('control'); sh.getRange('A1:B1').setValues([['', 0]]); } return sh; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function listSites_() { var rows = getSheet_().getDataRange().getValues(); var out = []; for (var i = 1; i < rows.length; i++) { if (!rows[i][1]) continue; out.push({ name: String(rows[i][0] || ''), siteUrl: String(rows[i][1]), repoUrl: String(rows[i][2] || ''), desc: String(rows[i][3] || '') }); } return out; } function readRun_() { var sh = getControl_(); return { url: String(sh.getRange('A1').getValue() || ''), at: Number(sh.getRange('B1').getValue() || 0) }; } // Read the list and the current Run command (needs the view password) function doGet(e) { var pw = (e && e.parameter && e.parameter.pw) || ''; if (pw !== VIEW_PW) return json_({ ok: false, error: 'wrong password' }); return json_({ ok: true, sites: listSites_(), run: readRun_() }); } // Add, remove, run, or verify (needs the admin password) function doPost(e) { var body = {}; try { body = JSON.parse(e.postData.contents); } catch (err) {} if (body.adminPw !== ADMIN_PW) return json_({ ok: false, error: 'wrong admin password' }); if (body.action === 'verify') { return json_({ ok: true, verified: true }); } var sh = getSheet_(); if (body.action === 'add' && body.site && body.site.siteUrl) { sh.appendRow([body.site.name || '', body.site.siteUrl, body.site.repoUrl || '', body.site.desc || '']); return json_({ ok: true, sites: listSites_() }); } if (body.action === 'remove' && body.siteUrl) { var rows = sh.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][1] === body.siteUrl) sh.deleteRow(i + 1); } return json_({ ok: true, sites: listSites_() }); } if (body.action === 'run' && body.siteUrl) { var now = Date.now(); getControl_().getRange('A1:B1').setValues([[body.siteUrl, now]]); return json_({ ok: true, at: now }); } return json_({ ok: false, error: 'unknown action' }); }
